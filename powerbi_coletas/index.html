Coletas B2C vs B2B = 
Coletas B2C = 
CALCULATE(
    [Total Coletas],
    dCanalCliente[TipoCanal] = "B2C"
)

Coletas B2B = 
CALCULATE(
    [Total Coletas],
    dCanalCliente[TipoCanal] = "B2B"

Coletores Necessários vs c/ Agenda = 
Coletores Necessarios = 
ROUNDUP(
    DIVIDE(
        [Total Coletas],
        [Capacidade Media Coletor Turno],
        0
    ),
    0
)

Indice Cobertura Escala = 
DIVIDE([Coletores c/ Agenda], [Coletores Necessarios], 0)

    [Total Coletas],
